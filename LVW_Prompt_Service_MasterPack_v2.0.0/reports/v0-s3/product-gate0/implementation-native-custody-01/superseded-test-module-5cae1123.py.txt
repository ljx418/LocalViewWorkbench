"""The offline validation of the N1 parent owned native custody.

The validation of this package is *offline*.  It never starts a process,
it never opens a socket, it never touches a database and it never reads
the original budget ledger or the original shared counter.  The only
native capability that is exercised here is an **in-process façade** of the
Windows primitives: a typed object with the same methods as the
production façade, which records the calls in their order and can fail at
one named call.

The guard of the process is installed *before* pytest, before any plugin
and before any import of the candidate modules.  It is an audit hook, so
the refusal of the operating system events happens inside the event
itself and before the action.  The installation is only promised after a
handshake proved that the hook of this process is really active.

The production entry of this module is::

    python3 -B tools/test_v0_s3_native_custody.py --run <nodeid> [<nodeid> ...]

The entry installs the guard, runs the selected nodes, writes the raw
logs and the receipt of the run and reports the true exit code of the
test process.
"""

import queue

# -- the guard of the process ----------------------------------------------

GUARD_REGISTRY_MODULE = 'lvw_s3_native_custody_guard_registry'
GUARD_HANDSHAKE_EVENT = 'lvw.s3.native-custody.guard.handshake'

# The events that would cross the offline boundary.  Every one of them is
# raised by the interpreter *before* the action, so a refusal of one of
# them is a proven position before the action.
BLOCKED_EVENTS = (
    'socket.connect',
    'socket.connect_ex',
    'socket.bind',
    'socket.getaddrinfo',
    'socket.gethostbyname',
    'socket.gethostbyaddr',
    'socket.getnameinfo',
    'socket.sendto',
    'subprocess.Popen',
    'os.system',
    'os.exec',
    'os.spawn',
    'os.posix_spawn',
    'os.fork',
    'os.forkpty',
    'sqlite3.connect',
)

# The original ledger of the product and the original shared counter of
# the finite CPU budget.  The ledger may not be read at all and the
# counter may only be read, outside the guarded batch, and only for the
# observation of the batch.
FORBIDDEN_LEDGER = ('/mnt/c/workspace/videoworkline/.lvw-runtime/lvw-prompt-service/'
                     'program02.sqlite')
FORBIDDEN_COUNTER = '/tmp/lvw-s3-p4-cpu-control-xn4cjn_s/shared-counter.json'


class OfflineBoundaryViolation(RuntimeError):
    """Raised when the offline boundary would be crossed."""


class GuardInstallationError(RuntimeError):
    """The offline guard of this process is not proven active."""

    def __init__(self, code, detail=None):
        super().__init__(code)
        self.code = code
        self.detail = detail


class _GuardRegistry:
    """The one, process wide owner of the offline guard."""

    def __init__(self):
        self.owner_id = None
        self.handshake_token = None
        self.hook_registered = False
        self.hook_installed = False
        self.hook_install_count = 0
        self.install_attempts = 0
        self.handshake_count = 0
        self.install_error = None
        self.refusals = []
        self.counter_read_allowed = False


def _guard_registry():
    registry = sys.modules.get(GUARD_REGISTRY_MODULE)
    if registry is None:
        registry = _GuardRegistry()
        sys.modules[GUARD_REGISTRY_MODULE] = registry
    return registry


def _is_forbidden_path(path):
    try:
        value = os.fspath(path)
    except TypeError:
        return False
    if isinstance(value, bytes):
        value = value.decode('utf-8', 'replace')
    if not isinstance(value, str):
        return False
    if not any(marker in value for marker in ('program02.sqlite', 'shared-counter.json')):
        return False
    if value in (FORBIDDEN_LEDGER, FORBIDDEN_COUNTER):
        return True
    try:
        resolved = str(Path(value).resolve())
    except (OSError, ValueError):
        return False
    return resolved in (FORBIDDEN_LEDGER, FORBIDDEN_COUNTER)


def _is_read_only(mode, flags):
    if isinstance(mode, str) and any(character in mode for character in 'wax+'):
        return False
    if isinstance(flags, int):
        blocked = (os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC | os.O_APPEND)
        if flags & blocked:
            return False
    return True


def _audit_hook(event, arguments):
    """The one audit hook of this process."""
    registry = _guard_registry()
    if event == GUARD_HANDSHAKE_EVENT:
        if (registry.owner_id and len(arguments) == 2
                and arguments[0] == registry.owner_id
                and arguments[1] == registry.handshake_token):
            registry.handshake_count += 1
        return
    if event == 'open':
        path = arguments[0] if arguments else None
        mode = arguments[1] if len(arguments) > 1 else None
        flags = arguments[2] if len(arguments) > 2 else None
        if _is_forbidden_path(path):
            if not registry.counter_read_allowed:
                registry.refusals.append('OFFLINE_BOUNDARY:open:' + str(path))
                raise OfflineBoundaryViolation('offline boundary violation: open '
                                               + repr(path))
            if not _is_read_only(mode, flags):
                registry.refusals.append('OFFLINE_BOUNDARY:write:' + str(path))
                raise OfflineBoundaryViolation('offline boundary violation: write '
                                               + repr(path))
        return
    if event in BLOCKED_EVENTS:
        registry.refusals.append('OFFLINE_BOUNDARY:' + event)
        raise OfflineBoundaryViolation('offline boundary violation: ' + event
                                       + ' ' + repr(arguments))


def confirm_guard_active():
    """Prove that the audit hook of this process is really active."""
    registry = _guard_registry()
    if registry.owner_id is None:
        return False
    before = registry.handshake_count
    try:
        sys.audit(GUARD_HANDSHAKE_EVENT, registry.owner_id, registry.handshake_token)
    except Exception:  # noqa: BLE001 - a failed handshake proves nothing
        return False
    return registry.handshake_count > before


def install_offline_guard():
    """Install the offline guard of this process, exactly once."""
    registry = _guard_registry()
    registry.install_attempts += 1
    if registry.hook_registered:
        if not confirm_guard_active():
            registry.hook_installed = False
            registry.install_error = 'GUARD_HANDSHAKE_FAILED'
            raise GuardInstallationError('GUARD_HANDSHAKE_FAILED')
        registry.hook_installed = True
        if registry.hook_install_count == 0:
            registry.hook_install_count = 1
        return registry.owner_id
    registry.owner_id = 'owner-' + hashlib.sha256(
        str(Path(__file__).resolve()).encode('utf-8')).hexdigest()[:32]
    registry.handshake_token = hashlib.sha256(
        ('%s|%s|%s' % (registry.owner_id, os.getpid(), time.time())).encode('utf-8')
    ).hexdigest()[:32]
    registry.hook_registered = True
    try:
        sys.addaudithook(_audit_hook)
    except Exception as error:  # noqa: BLE001 - the refusal is the guard
        registry.owner_id = None
        registry.hook_installed = False
        registry.install_error = 'GUARD_REGISTRATION_FAILED:' + type(error).__name__
        raise GuardInstallationError('GUARD_REGISTRATION_FAILED', str(error)) from error
    if not confirm_guard_active():
        registry.hook_installed = False
        registry.install_error = 'GUARD_HANDSHAKE_FAILED'
        raise GuardInstallationError('GUARD_HANDSHAKE_FAILED')
    registry.hook_installed = True
    registry.hook_install_count += 1
    return registry.owner_id

class FakeWin32:
    """The in-process façade of the Windows primitives.

    The façade has the same methods as the production façade.  It records
    the calls in their order, it can fail at one named call and it never
    creates a process, never opens a socket and never touches a file: it
    is a pure in-process object, and the assertions of this package are
    made on its record, not on a claim of the implementation.
    """

    def __init__(self, *, fail_at=None, job_process_ids=None, stream_payload=None):
        self.fail_at = fail_at
        self.calls = []
        self.closed = []
        self.terminated = []
        self.assigned = []
        self.resumed = []
        self.created = []
        self.inherited = []
        self._job_ids = list(job_process_ids) if job_process_ids is not None else []
        self._stream_payload = dict(stream_payload or {})
        self._streams = {}
        self._next_handle = 100
        self._next_pid = 4000

    # -- helpers -----------------------------------------------------------
    def _handle(self):
        self._next_handle += 1
        return self._next_handle

    def _fail(self, name):
        if self.fail_at == name:
            raise BrokerError('BROKER_FAKE_FAILURE_' + name.upper())

    def names(self):
        return [call[0] for call in self.calls]

    # -- the job -----------------------------------------------------------
    def create_job(self):
        self.calls.append(('create_job',))
        self._fail('create_job')
        return self._handle()

    def set_job_kill_on_close(self, job):
        self.calls.append(('set_job_kill_on_close', job))
        self._fail('job_limit')

    def terminate_job(self, job, exit_code):
        self.calls.append(('terminate_job', job, exit_code))
        self._fail('terminate_job')
        for entry in self.created:
            entry['exited'] = True

    def job_process_ids(self, job):
        self.calls.append(('job_process_ids', job))
        self._fail('query_job')
        if len(self._job_ids) > 17:
            raise BrokerError('BROKER_DESCENDANT_LIMIT', 'count=%d' % len(self._job_ids))
        return list(self._job_ids)

    # -- the processes -----------------------------------------------------
    def create_process_suspended(self, *, executable, arguments, working_directory,
                                  stdout_handle, stderr_handle, environment=None):
        self.calls.append(('create_process_suspended', executable, tuple(arguments)))
        self._fail('create')
        self._next_pid += 1
        entry = {'pid': self._next_pid, 'exited': False, 'handle': self._handle(),
                 'thread': self._handle()}
        self.created.append(entry)
        self.inherited.append((stdout_handle, stderr_handle))
        return CreatedProcess(entry['handle'], entry['thread'], entry['pid'],
                              creation_time=17000000000000000)

    def assign_process_to_job(self, job, process):
        self.calls.append(('assign_process_to_job', job, process))
        self._fail('assign')
        self.assigned.append((job, process))

    def resume_thread(self, thread):
        self.calls.append(('resume_thread', thread))
        self._fail('resume')
        self.resumed.append(thread)

    def terminate_process(self, process, exit_code):
        self.calls.append(('terminate_process', process, exit_code))
        self._fail('terminate')
        self.terminated.append(process)
        for entry in self.created:
            if entry['handle'] == process:
                entry['exited'] = True

    def wait_process(self, process, timeout_ms):
        self.calls.append(('wait_process', process, timeout_ms))
        for entry in self.created:
            if entry['handle'] == process:
                return entry['exited']
        return True

    def exit_code(self, process):
        self.calls.append(('exit_code', process))
        return 0

    def close_handle(self, handle):
        self.calls.append(('close_handle', handle))
        self._fail('close')
        self.closed.append(handle)

    # -- the pipes ---------------------------------------------------------
    def create_pipe(self, name):
        self.calls.append(('create_pipe', name))
        self._fail('pipe')
        read = self._handle()
        write = self._handle()
        self._streams[read] = name
        self._streams[write] = name + '-write'
        return read, write

    def read_stream(self, handle, size):
        payload = self._stream_payload.get(self._streams.get(handle, ''), b'')
        if not payload:
            return b''
        self._stream_payload[self._streams.get(handle, '')] = b''
        return payload

    def write_stream(self, handle, data):
        self.calls.append(('write_stream', handle, len(data)))


class MemoryChannel:
    """The in-process transport of the private control channel.

    The transport is a pair of queues.  It never opens a pipe, a socket or
    a file and it never starts a process.
    """

    def __init__(self):
        self.inbound = queue.Queue()
        self.outbound = []
        self.closed = False
        self.eof = False

    def read(self, timeout):
        try:
            return self.inbound.get(timeout=max(0.0, timeout))
        except queue.Empty:
            return b'' if self.eof else None

    def write(self, raw):
        self.outbound.append(raw)

    def close(self):
        self.closed = True


class FakeBrokerProcess:
    """The in-process stand-in of the original handle of the broker.

    The stand-in speaks the closed frame protocol of the private channel.
    It is not a simulation of a Windows process: it is a queue of frames,
    and the assertions of this package are made on the frames themselves.
    """

    def __init__(self, plan, nonce, kinds=('READY', 'STATUS', 'STOPPED')):
        self.plan = plan
        self.nonce = nonce
        self.pid = 4242
        self.executable = 'C:/Windows/py.exe'
        self.executable_sha256 = 'f' * 64
        self.inbound = queue.Queue()
        self.outbound = []
        self.closed = False
        self.close_count = 0
        states = {'READY': 'ARMED', 'STATUS': 'RUNNING', 'STOPPED': 'RELEASED'}
        for index, kind in enumerate(kinds, start=1):
            self.inbound.put(_reply_frame(kind, states[kind], {}, plan, nonce, index))

    def write(self, raw):
        self.outbound.append(raw)

    def read(self, timeout):
        try:
            return self.inbound.get(timeout=max(0.0, timeout))
        except queue.Empty:
            return None

    def close(self):
        self.closed = True
        self.close_count += 1


class Clock:
    """A clock that only moves when the test moves it."""

    def __init__(self, value):
        self.value = float(value)

    def __call__(self):
        return self.value

    def advance(self, seconds):
        self.value += float(seconds)

def _reply_frame(kind, state, detail, plan, nonce, sequence):
    return json.dumps({'schema': CHANNEL_SCHEMA, 'kind': kind, 'run_id': plan.run_id,
                       'candidate_sha': plan.candidate_sha,
                       'authorization_sha256': plan.authorization_sha256,
                       'nonce': nonce, 'sequence': sequence, 'state': state,
                       'detail': detail},
                      ensure_ascii=False, sort_keys=True,
                      separators=(',', ':')).encode('utf-8') + b'\n'

    return NativeCustodySession(plan=document, run_root=tmp_path,
                                 parent_child_remaining_seconds=remaining,
                                 launcher=launcher, clock=clock, wall=wall), state

def _counter_snapshot():
    registry = _guard_registry()
    path = Path(FORBIDDEN_COUNTER)
    try:
        registry.counter_read_allowed = True
        raw = path.read_bytes()
    except OSError as error:
        return {'path': str(path), 'readable': False, 'error_type': type(error).__name__,
                'written_by_this_run': False, 'refunded_by_this_run': False,
                'reset_by_this_run': False}
    finally:
        registry.counter_read_allowed = False
    return {'path': str(path), 'readable': True,
            'sha256': hashlib.sha256(raw).hexdigest(), 'bytes': len(raw),
            'written_by_this_run': False, 'refunded_by_this_run': False,
            'reset_by_this_run': False}
