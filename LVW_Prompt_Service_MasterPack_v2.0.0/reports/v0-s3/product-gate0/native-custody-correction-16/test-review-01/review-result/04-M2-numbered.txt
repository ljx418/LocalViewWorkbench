 4900         """Cancel one worker and prove that it really stopped."""
 4901         state = worker.request_cancel(self._effective_deadline(deadline))
 4902         if not worker.join_bounded(self._remaining(deadline)):
 4903             # The worker did not stop: it keeps its owner reference and

 4910                               '%s:%s' % (self._name, state))

 4914             raise BrokerError('BROKER_PIPE_CANCEL_NOT_CONFIRMED', self._name)
