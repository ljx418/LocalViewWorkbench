  820

  836         try:
  837             while not self._stop_event.is_set():
  838                 remaining = self._remaining()

  840                     # The bound of the round arrived: the pump asks for
  841                     # nothing more and it ends.  It never waits with a

  869     def _deliver(self, reply):
