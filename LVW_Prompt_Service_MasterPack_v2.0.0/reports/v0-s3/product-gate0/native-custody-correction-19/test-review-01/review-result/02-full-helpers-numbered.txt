  406 class _PhaseLedger:
  407     """The ledger of the phases that the group of this module runs.
  408 
  409     The ledger remembers what was **entered** and what **ended**.  It
  410     never decides: a phase that was entered and that failed is a
  411     ``FAIL`` and only a phase that was never entered is a ``NOT_RUN``.
  412     The states are ``PLANNED``, ``ATTEMPTED``, ``PASSED``, ``FAIL`` and
  413     ``NOT_RUN`` and the two last states are terminal.
  414     """
  415 
  416     STATES = ('PLANNED', 'ATTEMPTED', 'PASSED', 'FAIL', 'NOT_RUN')
  417 
  418     def __init__(self):
  419         self._lock = threading.Lock()
  420         self._states = {}
  421         self._order = []
  422         self._reasons = {}
  423 
  424     def _plan_locked(self, name):
  425         if name not in self._states:
  426             self._states[name] = 'PLANNED'
  427             self._order.append(name)
  428 
  429     def plan(self, name):
  430         """Name one phase **before** it runs."""
  431         with self._lock:
  432             self._plan_locked(name)
  433             return name

  435     def attempt(self, name):
  436         """Mark that the operation of the phase was **entered**."""
  437         with self._lock:
  438             self._plan_locked(name)
  439             if self._states[name] == 'PLANNED':
  440                 self._states[name] = 'ATTEMPTED'
  441             return self._states[name]

  443     def finish(self, name, state, reason=None):
  444         """Close one phase; a closed phase is never reopened."""
  445         with self._lock:
  446             self._plan_locked(name)
  447             if self._states[name] in ('PASSED', 'FAIL', 'NOT_RUN'):
  448                 return self._states[name]

  450             if reason is not None:
  451                 self._reasons[name] = reason
  452             return state

  456         return self.finish(name, 'PASSED')

  458     def fail(self, name, reason=None):

  460         return self.finish(name, 'FAIL', reason)

  462     def state(self, name):
  463         """The state of one phase."""
  464         with self._lock:
  465             return self._states.get(name, 'PLANNED')

  513 def _owner_start_state(thread, hint=None):
  514     """The start state of one owner of the round.
  515 
  516     The state is one of ``'STARTED'``, ``'NEVER_STARTED'`` or
  517     ``'UNKNOWN'``.  ``UNKNOWN`` is the state of an owner whose start
  518     did not return and of an owner whose liveness could not be asked:
  519     such an owner may still come up, it is never taken for an owner
  520     that never started and it **fails the proof** of the cleanup.
  521 
  522     CORRECTION 17 (M3): ``NEVER_STARTED`` is returned **only** when the
  523     provenance of the start **proves** it: the owner does not exist at
  524     all, or the registry of the fixture observed that its start was
  525     never invoked.  An owner without an identity and without a
  526     provenance is ``UNKNOWN``: it is never guessed to be an owner that
  527     never started.
  528     """
  529     if thread is None:
  530         return 'NEVER_STARTED'
  531     # CORRECTION 18 (M3): the query of the identity of the owner is
  532     # **not** swallowed here any more.  A failure of it is a fact of the
  533     # cleanup and the guard of the caller records it with its exact type
  534     # and its exact detail: a silent ``UNKNOWN`` without a diagnostic
  535     # would hide the failure of the classification and it would let an
  536     # unknown identity pass for a proved exit.
  537     ident = getattr(thread, 'ident', None)
  538     if ident is not None:
  539         return 'STARTED'
  540     if hint in ('STARTED', 'NEVER_STARTED', 'UNKNOWN'):
  541         return hint
  542     return 'UNKNOWN'

  545 def _salvage_threads(owners, *, seconds=10.0, states=None, total_seconds=30.0,
  546                      interrupts=None):
  547     """Join the owners that really started and report their liveness.
  548 
  549     The salvage is the work of the **test** and it is **separate** from
  550     the returned state of the product: it never renews a deadline of
  551     the round, it never erases a retained resource of the product and
  552     it never claims that the product released it.
  553 
  554     Every owner that really started is joined with an explicitly
  555     bounded wait and its liveness is reported.  An owner that never
  556     started is reported and **never** joined.  An owner whose start
  557     state is ``UNKNOWN`` is reported as unknown: it is never joined as
  558     if it had ended and it fails the proof of the cleanup.
  559 
  560     CORRECTION 17 (M3): **every** owner is attempted.  A failure of the
  561     classification of an owner, of its join or of its liveness query
  562     never aborts the iteration: the error is **retained** in a list
  563     with its kind, its exact type and its detail, and it fails the
  564     proof of the cleanup.  The wait of an owner is cropped by the
  565     remaining budget of the salvage, so that the salvage is bounded as
  566     a whole.
  567 
  568     CORRECTION 18 (M3): the **exact** object of a ``KeyboardInterrupt``
  569     or of a ``SystemExit`` is retained in the ``interrupts`` list of the
  570     caller: the identity of the cancellation survives every best effort
  571     cleanup and it is re-raised by the owner of the cleanup when no
  572     other failure is unwinding.  The error that is written in the
  573     document keeps its exact type and its exact detail; the **object**
  574     itself never enters a document.
  575     """
  576     facts = []
  577     hints = states or {}
  578     deadline = time.monotonic() + max(0.0, total_seconds)
  579     for name, thread in owners:
  580         state = None
  581         errors = []
  582         try:
  583             state = _owner_start_state(thread, hints.get(name))
  584         except BaseException as error:  # noqa: BLE001 - it stays visible
  585             errors.append({'kind': 'STATE', 'type': type(error).__name__,
  586                            'detail': str(error)})
  587             _note_interruption(interrupts, error)
  588             state = 'UNKNOWN'
  589         if state != 'STARTED':
  590             facts.append({'name': name, 'start_state': state,
  591                           'started': None if state == 'UNKNOWN' else False,
  592                           'alive': None, 'joined': None, 'errors': errors})
  593             continue
  594         try:
  595             wait = max(0.0, min(seconds, deadline - time.monotonic()))
  596             thread.join(wait)
  597         except BaseException as error:  # noqa: BLE001 - it stays visible
  598             errors.append({'kind': 'JOIN', 'type': type(error).__name__,
  599                            'detail': str(error)})
  600             _note_interruption(interrupts, error)
  601         alive = None
  602         try:
  603             alive = bool(thread.is_alive())
  604         except BaseException as error:  # noqa: BLE001 - it stays visible
  605             errors.append({'kind': 'LIVENESS', 'type': type(error).__name__,
  606                            'detail': str(error)})
  607             _note_interruption(interrupts, error)
  608         facts.append({'name': name, 'start_state': state, 'started': True,
  609                       'alive': alive, 'joined': (alive is False),
  610                       'errors': errors})
  611     return facts

  644 def _assert_salvage_proved(facts, message):
  645     """The proof that the fixture released every owner that it started.
  646 
  647     An owner with an **unknown** start state fails the proof: it is
  648     never taken for an owner that never started and it is never taken

  658             '%s: the owner %s met %s' % (message, item['name'], errors)

  661                 '%s: the owner %s was left behind' % (message, item['name'])

  664 def _prove_salvage(facts, message):
  665     """The proof of the cleanup of a round, as a **fact**.
  666 
  667     The proof is **run** in every case - also while an ordinary failure
  668     is already unwinding - and it returns the failure of the proof
  669     instead of raising it: a failure of the proof never replaces the
  670     failure that is already unwinding and it is never discarded.
  671     """
  672     for item in facts:
  673         errors = item.get('errors') or []
  674         if item['start_state'] == 'UNKNOWN':

  680         if item['started'] and item['alive'] is not False:
  681             return '%s: the owner %s was left behind' % (message, item['name'])
  682     return None

  739         captured['evidence_state'] = evidence['state']
  740         captured['evidence_broker_retained'] = evidence['broker_retained']

  743             evidence['background_threads'])
  744         captured['evidence_closer_start_state'] = evidence['closer_start_state']
  745     except BaseException as error:  # noqa: BLE001 - the failure stays visible

  747                                       'detail': str(error)}

  782 def _cleanup_detail(cleanup_errors, write_errors):
  783     """The diagnostic of the cleanup errors of one phase.
  784

  800 def _session_owners(session, prefix='session', hints=None):
  801     """The owners that the session of the round really started.
  802 
  803     The registry is read from the session itself: the background
  804     threads of the round and the closer threads that it really built.
  805     A closer that never started is reported as such and it is never
  806     joined.  A thread whose start did not return is **UNKNOWN**: it
  807     keeps its owner in the session and it may still come up.
  808     """
  809     owners = []
  810     if session is None:
  811         return owners
  812     records = getattr(session, '_background_threads', None) or {}
  813     for kind in sorted(records):
  814         record = records[kind] or {}
  815         name = '%s:background-%s' % (prefix, str(kind))
  816         owners.append((name, record.get('thread')))
  817         if hints is None:
  818             continue
  819         state = record.get('state')
  820         if state == 'STARTING':
  821             hints[name] = 'UNKNOWN'
  822         elif state == 'FAILED' and not record.get('started'):
  823             hints[name] = 'NEVER_STARTED'
  824     closer_state = getattr(session, '_closer_start_state', None)
  825     for index, thread in enumerate(getattr(session, '_closer_threads', None)
  826                                    or []):
  827         name = '%s:closer-%d' % (prefix, index)
  828         owners.append((name, thread))
  829         if hints is not None and closer_state == 'STARTING':
  830             hints[name] = 'UNKNOWN'
  831     return owners

  834 def _endpoint_owners(prefix, endpoint, hints=None):
  835     """The owners that one endpoint of the round really started.
  836 
  837     The read worker and the write worker of the endpoint are the
  838     owners that the endpoint really started.  A worker that the
  839     endpoint **retained** is a worker whose start did not return: its
  840     state is **UNKNOWN** and it fails the proof of the cleanup.
  841     """
  842     owners = []
  843     if endpoint is None:
  844         return owners
  845     for role, label in (('_reader', 'read'), ('_writer', 'write')):
  846         worker = getattr(endpoint, role, None)
  847         if worker is None:
  848             continue
  849         owners.append(('%s:%s-worker' % (prefix, label), worker))
  850     for index, worker in enumerate(getattr(endpoint, '_retained', None)
  851                                    or []):
  852         name = '%s:retained-%d' % (prefix, index)
  853         owners.append((name, worker))
  854         if hints is not None:
  855             hints[name] = 'UNKNOWN'
  856     return owners

  859 def _channel_owners(channel, prefix, hints=None):
  860     """The owners that the channel of the round really started.
  861 
  862     The drain thread of the channel and the workers of its two
  863     endpoints are the owners of the round that really started.  A worker
  864     that was never started is reported as such and it is never joined.
  865     """
  866     owners = []
  867     if channel is None:
  868         return owners
  869     owners.append((prefix + ':drain', getattr(channel, '_thread', None)))
  870     if hints is not None:
  871         # The drain thread of a channel is started **inside** the
  872         # constructor of the channel: a channel whose construction
  873         # failed at the start of its drain owns a drain thread whose
  874         # start did not return and it may still come up.
  875         hints[prefix + ':drain'] = 'UNKNOWN'
  876     reader = getattr(channel, '_reader', None)
  877     writer = getattr(channel, '_writer', None)
  878     for endpoint, label in ((reader, 'read'), (writer, 'write')):
  879         owners.extend(_endpoint_owners('%s:%s' % (prefix, label), endpoint,
  880                                        hints))
  881     return owners

  884 def _registry_owners(registry, session=None, channel=None, prefix='round',
  885                       errors=None, interrupts=None):
  886     """Every owner of the round: the registered ones and the live ones.
  887 
  888     The registry of the fixture holds the owners that were registered
  889     **immediately** - a channel that was constructed, an endpoint that
  890     was built, a session that was created.  The session and the
  891     channel that are still reachable are added to them, so that a
  892     thread that the product started after the registration is never
  893     forgotten.  One owner is never in the list twice: the identity of
  894     the thread decides and not its name.
  895 
  896     CORRECTION 19 (M3): the traversal is **best effort**.  The query
  897     of one container never aborts the traversal and it never replaces
  898     the owners of the round with an empty list: every other owner is
  899     still attempted.  The **exact** thread objects that the start
  900     registry observed are retained **first** and they are an
  901     independent fallback - a failure of the query of a container never
  902     loses them.  Every failure is retained with its exact type and its
  903     exact detail and the **exact** object of an interruption is kept.
  904     """
  905     owners = []
  906     hints = {}
  907     seen = set()
  908     kinds = set()
  909 
  910     def add(items):
  911         for name, thread in items:
  912             key = ('thread', id(thread)) if thread is not None \
  913                 else ('name', name)
  914             if key in seen:
  915                 continue
  916             seen.add(key)
  917             owners.append((name, thread))
  918 
  919     def note(kind, error):
  920         """Retain one failure of the traversal, never swallow it."""
  921         if errors is not None:
  922             errors.append({'kind': kind, 'type': type(error).__name__,
  923                            'detail': str(error)})
  924         _note_interruption(interrupts, error)
  925 
  926     try:
  927         entries = registry.entries()
  928     except BaseException as error:  # noqa: BLE001 - it stays visible
  929         note('REGISTRY_ENTRIES', error)
  930         entries = []
  931     # CORRECTION 19 (M3): the **exact** thread objects that the start
  932     # registry observed are retained **first** and they are an
  933     # independent fallback: a failure of the query of a container never
  934     # loses them and it never replaces the owners of the round with an
  935     # empty list.  They are merged into the owners **after** the
  936     # traversal, so that the name of an owner that a container really
  937     # knows is never replaced by the name of the registry.
  938     observed = [(item.get('name'), item.get('owner'))
  939                 for item in entries
  940                 if item.get('kind') == 'thread'
  941                 and item.get('owner') is not None]
  942     for entry in entries:
  943         name = entry.get('name')
  944         owner = entry.get('owner')
  945         kind = entry.get('kind')
  946         if kind == 'thread':
  947             add([(name, owner)])
  948             continue
  949         try:
  950             if kind == 'channel':
  951                 items = _channel_owners(owner, name, hints)
  952             elif kind == 'endpoint':
  953                 items = _endpoint_owners(name, owner, hints)
  954             elif kind == 'session':
  955                 items = _session_owners(owner, name, hints)
  956             else:
  957                 items = [(name, owner)]
  958         except BaseException as error:  # noqa: BLE001 - it stays visible
  959             # the traversal continues: the owners that follow this one
  960             # are still attempted and the exact thread object of this
  961             # entry is still an owner of the round.
  962             note('OWNER_QUERY', error)
  963             if isinstance(owner, threading.Thread):
  964                 add([(name, owner)])
  965             continue
  966         kinds.add(kind)
  967         add(items)
  968     try:
  969         if session is not None and 'session' not in kinds:
  970             add(_session_owners(session, '%s:session' % prefix, hints))
  971     except BaseException as error:  # noqa: BLE001 - it stays visible
  972         note('SESSION_QUERY', error)
  973     try:
  974         if channel is not None and 'channel' not in kinds:
  975             add(_channel_owners(channel, '%s:channel' % prefix, hints))
  976     except BaseException as error:  # noqa: BLE001 - it stays visible
  977         note('CHANNEL_QUERY', error)
  978     # CORRECTION 19 (M3): the fallback of the observed threads: every
  979     # exact thread object that the start registry saw is an owner of the
  980     # round.  The merge is **idempotent**: an owner that the traversal
  981     # already named keeps its name and its hint.
  982     for name, thread in observed:
  983         add([(name, thread)])
  984     # CORRECTION 17 (M3): the provenance of the starts that the
  985     # observer of the fixture saw is the **proven** provenance of the
  986     # start of an owner.  It is keyed by the exact thread object and it
  987     # never replaces the hint of the production: a thread whose start
  988     # returned is ``STARTED`` and a thread whose start did not return is
  989     # ``UNKNOWN``.
  990     try:
  991         provenance = registry.start_provenance() if registry is not None else {}
  992     except BaseException as error:  # noqa: BLE001 - it stays visible
  993         note('PROVENANCE', error)
  994         provenance = {}
  995     for name, thread in owners:
  996         if thread is None:
  997             continue
  998         state = provenance.get(id(thread))
  999         if state == 'STARTED':
 1000             hints[name] = 'STARTED'
 1001         elif state in ('STARTING', 'FAILED'):
 1002             hints[name] = 'UNKNOWN'
 1003     return owners, hints

 1006 def _drain_owners(registry, session=None, channel=None, prefix='round', *,
 1007                   seconds=10.0, budget=30.0, release=None, interrupts=None):
 1008     """The bounded drain of the owners of one round.
 1009 
 1010     CORRECTION 17 (M3): the drain repeats, **boundedly**, until the
 1011     owners of the round are quiescent and really joined.  The bound of
 1012     the drain is **one** overall bound: the wait of an owner is cropped
 1013     by the remaining budget and the budget is never reset - neither
 1014     between the rounds of the drain nor between the rounds of the test.
 1015 
 1016     The quiescence of the round is asserted **after** the drain and
 1017     never before it: a round that is not quiescent is a failure of the
 1018     cleanup, the gates of the round are still released and **all** the
 1019     registered owners are still attempted.
 1020 
 1021     CORRECTION 18 (M3): the enumeration of the owners and the reading
 1022     of the generation of the registry are **one** observation.  The
 1023     generation is read **before** the exact owners are enumerated and
 1024     it is read **again** right after the enumeration and once more
 1025     after the joined producer threads of the iteration.  All of these
 1026     readings must agree and the **final** exact-owner resnapshot must
 1027     contain every owner of the enumeration: a thread that was started
 1028     between the enumeration and the reading of the generation is never
 1029     certified as quiescent and it is never left behind.
 1030 
 1031     Every error of the drain - the enumeration, the release of the
 1032     gates, the classification of an owner, its join and its liveness
 1033     query - is **retained** in a list with its exact type and its exact
 1034     detail and it fails the proof of the cleanup.  **Every** iteration
 1035     is kept: an error of an earlier iteration is never erased by a
 1036     later iteration that really ended the threads.  A failure never
 1037     aborts the iteration and it never skips the final result of the
 1038     round.  The drain never renews a deadline of the product.
 1039 
 1040     CORRECTION 19 (M3): every read of the drain - the generation, the
 1041     pending starts and the final resnapshot - is **best effort** and it
 1042     never skips the owners that follow it.  The release of the gates is
 1043     **one action per gate**: a failure of one gate never skips the gate
 1044     that follows it.  The traversal of the owners never replaces the
 1045     owners of the round with an empty list and the exact thread objects
 1046     that the start registry observed are always retained.
 1047     """
 1048     deadline = time.monotonic() + max(0.0, budget)
 1049     attempts = []
 1050     errors = []
 1051     owners = []
 1052     hints = {}
 1053     salvage = []
 1054     all_salvage = []
 1055     quiescent = False
 1056 
 1057     def read_generation(label):
 1058         """Read the generation of the registry, guarded.
 1059 
 1060         CORRECTION 19 (M3): the reading of the generation is **best
 1061         effort**: a failure of it is a fact of the cleanup and it never
 1062         skips the owners that follow it.
 1063         """
 1064         try:
 1065             return registry.generation()
 1066         except BaseException as error:  # noqa: BLE001 - it stays visible
 1067             errors.append({'kind': label, 'type': type(error).__name__,
 1068                            'detail': str(error)})
 1069             _note_interruption(interrupts, error)
 1070             return None
 1071 
 1072     def read_pending(label):
 1073         """Read the pending starts of the registry, guarded."""
 1074         try:
 1075             return registry.pending_starts()
 1076         except BaseException as error:  # noqa: BLE001 - it stays visible
 1077             errors.append({'kind': label, 'type': type(error).__name__,
 1078                            'detail': str(error)})
 1079             _note_interruption(interrupts, error)
 1080             return None
 1081 
 1082     while True:
 1083         # the generation of the registry is read **before** the exact
 1084         # owners are enumerated
 1085         generation_before = read_generation('GENERATION_BEFORE')
 1086         try:
 1087             owners, hints = _registry_owners(registry, session=session,
 1088                                               channel=channel, prefix=prefix,
 1089                                               errors=errors,
 1090                                               interrupts=interrupts)
 1091         except BaseException as error:  # noqa: BLE001 - it stays visible
 1092             errors.append({'kind': 'ENUMERATION',
 1093                            'type': type(error).__name__,
 1094                            'detail': str(error)})
 1095             _note_interruption(interrupts, error)
 1096             # CORRECTION 19 (M3): the owners of the previous observation
 1097             # are **retained**: a failure of the enumeration never loses
 1098             # them and it never replaces them with an empty list.
 1099         # ... and it is read **again** right after the enumeration: a
 1100         # start that happened between the two readings is a change of
 1101         # the generation and the enumeration is repeated.
 1102         generation_after = read_generation('GENERATION_AFTER')
 1103         snapshot = {'generation_before': generation_before,
 1104                     'generation_after': generation_after,
 1105                     'generation': generation_after,
 1106                     'pending': read_pending('PENDING'),
 1107                     'owners': [name for name, _ in owners],
 1108                     'errors': list(errors)}
 1109         attempts.append(snapshot)
 1110         if release is not None:
 1111             # CORRECTION 19 (M3): the release of the gates is **best
 1112             # effort**: every gate is released on its own and a failure
 1113             # of one gate never skips the gate that follows it.
 1114             actions = (release if isinstance(release, (list, tuple))
 1115                        else [release])
 1116             for action in actions:
 1117                 try:
 1118                     action()
 1119                 except BaseException as error:  # noqa: BLE001
 1120                     errors.append({'kind': 'RELEASE',
 1121                                    'type': type(error).__name__,
 1122                                    'detail': str(error)})
 1123                     _note_interruption(interrupts, error)
 1124         remaining = max(0.0, deadline - time.monotonic())
 1125         salvage = _salvage_threads(owners, states=hints, seconds=seconds,
 1126                                    total_seconds=remaining,
 1127                                    interrupts=interrupts)
 1128         # CORRECTION 18 (M3): **every** iteration of the drain is kept -
 1129         # the failure of an earlier iteration is never erased by a later
 1130         # iteration that ended the threads.
 1131         all_salvage.extend(salvage)
 1132         # The **final** exact-owner resnapshot: it is taken after the
 1133         # joined producer threads of this iteration and it must agree
 1134         # with the enumeration of this iteration.
 1135         final_generation = read_generation('FINAL_GENERATION')
 1136         try:
 1137             final_owners, _final_hints = _registry_owners(
 1138                 registry, session=session, channel=channel, prefix=prefix,
 1139                 errors=errors, interrupts=interrupts)
 1140         except BaseException as error:  # noqa: BLE001 - it stays visible
 1141             errors.append({'kind': 'FINAL_ENUMERATION',
 1142                            'type': type(error).__name__,
 1143                            'detail': str(error)})
 1144             _note_interruption(interrupts, error)
 1145             final_owners = []
 1146         # and the generation is read **again** after the final
 1147         # resnapshot: a start that happened during it is a change of the
 1148         # generation and the drain repeats.
 1149         final_generation_after = read_generation('FINAL_GENERATION_AFTER')
 1150         final_names = [name for name, _ in final_owners]
 1151         missing = [name for name in snapshot['owners']
 1152                    if name not in final_names]
 1153         late = [name for name in final_names
 1154                 if name not in snapshot['owners']]
 1155         pending = read_pending('FINAL_PENDING')
 1156         stable = (generation_before == generation_after
 1157                   and final_generation == generation_after
 1158                   and final_generation_after == generation_after
 1159                   and not missing and not late)
 1160         quiescent = (pending == 0 and stable
 1161                      and _prove_salvage(salvage, prefix) is None)
 1162         if quiescent or time.monotonic() >= deadline:
 1163             break
 1164         # the drain is **bounded** and it never spins: the next attempt
 1165         # follows a short pause and it happens inside the same budget
 1166         time.sleep(0.01)
 1167     proof = _prove_salvage(all_salvage, prefix)
 1168     if proof is None and errors:
 1169         proof = '%s: the drain met %s' % (prefix, errors)
 1170     # CORRECTION 19 (M2/M3): a round that is **not** quiescent is a
 1171     # failure of the cleanup even when every owner that the salvage saw
 1172     # was really joined: the final resnapshot of the drain found an
 1173     # owner that no join released, and the proof of the cleanup says so.
 1174     if proof is None and not quiescent:
 1175         proof = ('%s: the owners of the round were not quiescent' % prefix)
 1176     return {'owners': owners, 'hints': hints, 'salvage': salvage,
 1177             'all_salvage': all_salvage,
 1178             'attempts': attempts, 'proof': proof, 'errors': errors,
 1179             'quiescent': bool(quiescent)}
