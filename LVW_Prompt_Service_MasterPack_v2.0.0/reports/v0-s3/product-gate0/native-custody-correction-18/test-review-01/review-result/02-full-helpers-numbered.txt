510:     """The start state of one owner of the round.
511: 
512:     The state is one of ``'STARTED'``, ``'NEVER_STARTED'`` or
513:     ``'UNKNOWN'``.  ``UNKNOWN`` is the state of an owner whose start
514:     did not return and of an owner whose liveness could not be asked:
515:     such an owner may still come up, it is never taken for an owner
516:     that never started and it **fails the proof** of the cleanup.
517: 
518:     CORRECTION 17 (M3): ``NEVER_STARTED`` is returned **only** when the
519:     provenance of the start **proves** it: the owner does not exist at
520:     all, or the registry of the fixture observed that its start was
521:     never invoked.  An owner without an identity and without a
522:     provenance is ``UNKNOWN``: it is never guessed to be an owner that
523:     never started.
524:     """

526:         return 'NEVER_STARTED'
527:     # CORRECTION 18 (M3): the query of the identity of the owner is
528:     # **not** swallowed here any more.  A failure of it is a fact of the
529:     # cleanup and the guard of the caller records it with its exact type
530:     # and its exact detail: a silent ``UNKNOWN`` without a diagnostic
531:     # would hide the failure of the classification and it would let an
532:     # unknown identity pass for a proved exit.

534:     if ident is not None:
535:         return 'STARTED'
536:     if hint in ('STARTED', 'NEVER_STARTED', 'UNKNOWN'):

541: def _salvage_threads(owners, *, seconds=10.0, states=None, total_seconds=30.0,
542:                      interrupts=None):
543:     """Join the owners that really started and report their liveness.
544: 
545:     The salvage is the work of the **test** and it is **separate** from
546:     the returned state of the product: it never renews a deadline of
547:     the round, it never erases a retained resource of the product and
548:     it never claims that the product released it.
549: 
550:     Every owner that really started is joined with an explicitly
551:     bounded wait and its liveness is reported.  An owner that never
552:     started is reported and **never** joined.  An owner whose start
553:     state is ``UNKNOWN`` is reported as unknown: it is never joined as
554:     if it had ended and it fails the proof of the cleanup.
555: 
556:     CORRECTION 17 (M3): **every** owner is attempted.  A failure of the
557:     classification of an owner, of its join or of its liveness query
558:     never aborts the iteration: the error is **retained** in a list
559:     with its kind, its exact type and its detail, and it fails the
560:     proof of the cleanup.  The wait of an owner is cropped by the
561:     remaining budget of the salvage, so that the salvage is bounded as
562:     a whole.
563: 
564:     CORRECTION 18 (M3): the **exact** object of a ``KeyboardInterrupt``
565:     or of a ``SystemExit`` is retained in the ``interrupts`` list of the
566:     caller: the identity of the cancellation survives every best effort
567:     cleanup and it is re-raised by the owner of the cleanup when no
568:     other failure is unwinding.  The error that is written in the
569:     document keeps its exact type and its exact detail; the **object**
570:     itself never enters a document.
571:     """
572:     facts = []
573:     hints = states or {}
574:     deadline = time.monotonic() + max(0.0, total_seconds)
575:     for name, thread in owners:
576:         state = None
577:         errors = []
578:         try:
579:             state = _owner_start_state(thread, hints.get(name))
580:         except BaseException as error:  # noqa: BLE001 - it stays visible
581:             errors.append({'kind': 'STATE', 'type': type(error).__name__,
582:                            'detail': str(error)})
583:             _note_interruption(interrupts, error)
584:             state = 'UNKNOWN'
585:         if state != 'STARTED':
586:             facts.append({'name': name, 'start_state': state,
587:                           'started': None if state == 'UNKNOWN' else False,
588:                           'alive': None, 'joined': None, 'errors': errors})
589:             continue
590:         try:
591:             wait = max(0.0, min(seconds, deadline - time.monotonic()))
592:             thread.join(wait)
593:         except BaseException as error:  # noqa: BLE001 - it stays visible
594:             errors.append({'kind': 'JOIN', 'type': type(error).__name__,
595:                            'detail': str(error)})
596:             _note_interruption(interrupts, error)
597:         alive = None
598:         try:
599:             alive = bool(thread.is_alive())
600:         except BaseException as error:  # noqa: BLE001 - it stays visible
601:             errors.append({'kind': 'LIVENESS', 'type': type(error).__name__,
602:                            'detail': str(error)})
603:             _note_interruption(interrupts, error)
604:         facts.append({'name': name, 'start_state': state, 'started': True,
605:                       'alive': alive, 'joined': (alive is False),
606:                       'errors': errors})
607:     return facts

782: def _session_owners(session, prefix='session', hints=None):
783:     """The owners that the session of the round really started.
784: 
785:     The registry is read from the session itself: the background
786:     threads of the round and the closer threads that it really built.
787:     A closer that never started is reported as such and it is never
788:     joined.  A thread whose start did not return is **UNKNOWN**: it
789:     keeps its owner in the session and it may still come up.
790:     """
791:     owners = []
792:     if session is None:
793:         return owners
794:     records = getattr(session, '_background_threads', None) or {}
795:     for kind in sorted(records):
796:         record = records[kind] or {}
797:         name = '%s:background-%s' % (prefix, str(kind))
798:         owners.append((name, record.get('thread')))
799:         if hints is None:
800:             continue
801:         state = record.get('state')
802:         if state == 'STARTING':
803:             hints[name] = 'UNKNOWN'
804:         elif state == 'FAILED' and not record.get('started'):
805:             hints[name] = 'NEVER_STARTED'
806:     closer_state = getattr(session, '_closer_start_state', None)
807:     for index, thread in enumerate(getattr(session, '_closer_threads', None)
808:                                    or []):
809:         name = '%s:closer-%d' % (prefix, index)
810:         owners.append((name, thread))
811:         if hints is not None and closer_state == 'STARTING':
812:             hints[name] = 'UNKNOWN'
813:     return owners

841: def _channel_owners(channel, prefix, hints=None):
842:     """The owners that the channel of the round really started.
843: 
844:     The drain thread of the channel and the workers of its two
845:     endpoints are the owners of the round that really started.  A worker
846:     that was never started is reported as such and it is never joined.
847:     """
848:     owners = []
849:     if channel is None:
850:         return owners
851:     owners.append((prefix + ':drain', getattr(channel, '_thread', None)))
852:     if hints is not None:
853:         # The drain thread of a channel is started **inside** the
854:         # constructor of the channel: a channel whose construction
855:         # failed at the start of its drain owns a drain thread whose
856:         # start did not return and it may still come up.
857:         hints[prefix + ':drain'] = 'UNKNOWN'
858:     reader = getattr(channel, '_reader', None)
859:     writer = getattr(channel, '_writer', None)
860:     for endpoint, label in ((reader, 'read'), (writer, 'write')):
861:         owners.extend(_endpoint_owners('%s:%s' % (prefix, label), endpoint,
862:                                        hints))

866: def _registry_owners(registry, session=None, channel=None, prefix='round'):
867:     """Every owner of the round: the registered ones and the live ones.
868: 
869:     The registry of the fixture holds the owners that were registered
870:     **immediately** - a channel that was constructed, an endpoint that
871:     was built, a session that was created.  The session and the
872:     channel that are still reachable are added to them, so that a
873:     thread that the product started after the registration is never
874:     forgotten.  One owner is never in the list twice: the identity of
875:     the thread decides and not its name.
876:     """
877:     owners = []
878:     hints = {}
879:     seen = set()
880:     kinds = set()
881: 
882:     def add(items):
883:         for name, thread in items:
884:             key = ('thread', id(thread)) if thread is not None \
885:                 else ('name', name)
886:             if key in seen:
887:                 continue
888:             seen.add(key)
889:             owners.append((name, thread))
890: 
891:     for entry in registry.entries():
892:         name = entry['name']
893:         owner = entry['owner']
894:         kind = entry['kind']
895:         kinds.add(kind)
896:         if kind == 'channel':
897:             add(_channel_owners(owner, name, hints))
898:         elif kind == 'endpoint':
899:             add(_endpoint_owners(name, owner, hints))
900:         elif kind == 'session':
901:             add(_session_owners(owner, name, hints))
902:         else:
903:             add([(name, owner)])
904:     if session is not None and 'session' not in kinds:
905:         add(_session_owners(session, '%s:session' % prefix, hints))
906:     if channel is not None and 'channel' not in kinds:
907:         add(_channel_owners(channel, '%s:channel' % prefix, hints))
908:     # CORRECTION 17 (M3): the provenance of the starts that the
909:     # observer of the fixture saw is the **proven** provenance of the
910:     # start of an owner.  It is keyed by the exact thread object and it
911:     # never replaces the hint of the production: a thread whose start
912:     # returned is ``STARTED`` and a thread whose start did not return is
913:     # ``UNKNOWN``.
914:     provenance = registry.start_provenance() if registry is not None else {}
915:     for name, thread in owners:
916:         if thread is None:
917:             continue
918:         state = provenance.get(id(thread))
919:         if state == 'STARTED':
920:             hints[name] = 'STARTED'
921:         elif state in ('STARTING', 'FAILED'):
922:             hints[name] = 'UNKNOWN'
923:     return owners, hints

926: def _drain_owners(registry, session=None, channel=None, prefix='round', *,
927:                   seconds=10.0, budget=30.0, release=None, interrupts=None):
928:     """The bounded drain of the owners of one round.
929: 
930:     CORRECTION 17 (M3): the drain repeats, **boundedly**, until the
931:     owners of the round are quiescent and really joined.  The bound of
932:     the drain is **one** overall bound: the wait of an owner is cropped
933:     by the remaining budget and the budget is never reset - neither
934:     between the rounds of the drain nor between the rounds of the test.
935: 
936:     The quiescence of the round is asserted **after** the drain and
937:     never before it: a round that is not quiescent is a failure of the
938:     cleanup, the gates of the round are still released and **all** the
939:     registered owners are still attempted.
940: 
941:     CORRECTION 18 (M3): the enumeration of the owners and the reading
942:     of the generation of the registry are **one** observation.  The
943:     generation is read **before** the exact owners are enumerated and
944:     it is read **again** right after the enumeration and once more
945:     after the joined producer threads of the iteration.  All of these
946:     readings must agree and the **final** exact-owner resnapshot must
947:     contain every owner of the enumeration: a thread that was started
948:     between the enumeration and the reading of the generation is never
949:     certified as quiescent and it is never left behind.
950: 
951:     Every error of the drain - the enumeration, the release of the
952:     gates, the classification of an owner, its join and its liveness
953:     query - is **retained** in a list with its exact type and its exact
954:     detail and it fails the proof of the cleanup.  **Every** iteration
955:     is kept: an error of an earlier iteration is never erased by a
956:     later iteration that really ended the threads.  A failure never
957:     aborts the iteration and it never skips the final result of the
958:     round.  The drain never renews a deadline of the product.
959:     """
960:     deadline = time.monotonic() + max(0.0, budget)
961:     attempts = []
962:     errors = []
963:     owners = []
964:     hints = {}
965:     salvage = []
966:     all_salvage = []
967:     quiescent = False
968:     while True:
969:         # the generation of the registry is read **before** the exact
970:         # owners are enumerated
971:         generation_before = registry.generation()
972:         try:
973:             owners, hints = _registry_owners(registry, session=session,
974:                                               channel=channel, prefix=prefix)
975:         except BaseException as error:  # noqa: BLE001 - it stays visible
976:             errors.append({'kind': 'ENUMERATION',
977:                            'type': type(error).__name__,
978:                            'detail': str(error)})
979:             owners, hints = [], {}
980:         # ... and it is read **again** right after the enumeration: a
981:         # start that happened between the two readings is a change of
982:         # the generation and the enumeration is repeated.
983:         generation_after = registry.generation()
984:         snapshot = {'generation_before': generation_before,
985:                     'generation_after': generation_after,
986:                     'generation': generation_after,
987:                     'pending': registry.pending_starts(),
988:                     'owners': [name for name, _ in owners],
989:                     'errors': list(errors)}
990:         attempts.append(snapshot)
991:         if release is not None:
992:             try:
993:                 release()
994:             except BaseException as error:  # noqa: BLE001 - it stays visible
995:                 errors.append({'kind': 'RELEASE',
996:                                'type': type(error).__name__,
997:                                'detail': str(error)})
998:         remaining = max(0.0, deadline - time.monotonic())
999:         salvage = _salvage_threads(owners, states=hints, seconds=seconds,
1000:                                    total_seconds=remaining,
1001:                                    interrupts=interrupts)
1002:         # CORRECTION 18 (M3): **every** iteration of the drain is kept -
1003:         # the failure of an earlier iteration is never erased by a later
1004:         # iteration that ended the threads.
1005:         all_salvage.extend(salvage)
1006:         # The **final** exact-owner resnapshot: it is taken after the
1007:         # joined producer threads of this iteration and it must agree
1008:         # with the enumeration of this iteration.
1009:         final_generation = registry.generation()
1010:         try:
1011:             final_owners, _final_hints = _registry_owners(
1012:                 registry, session=session, channel=channel, prefix=prefix)
1013:         except BaseException as error:  # noqa: BLE001 - it stays visible
1014:             errors.append({'kind': 'FINAL_ENUMERATION',
1015:                            'type': type(error).__name__,
1016:                            'detail': str(error)})
1017:             final_owners = []
1018:         # and the generation is read **again** after the final
1019:         # resnapshot: a start that happened during it is a change of the
1020:         # generation and the drain repeats.
1021:         final_generation_after = registry.generation()
1022:         final_names = [name for name, _ in final_owners]
1023:         missing = [name for name in snapshot['owners']
1024:                    if name not in final_names]
1025:         late = [name for name in final_names
1026:                 if name not in snapshot['owners']]
1027:         pending = registry.pending_starts()
1028:         stable = (generation_before == generation_after
1029:                   and final_generation == generation_after
1030:                   and final_generation_after == generation_after
1031:                   and not missing and not late)
1032:         quiescent = (pending == 0 and stable
1033:                      and _prove_salvage(salvage, prefix) is None)
1034:         if quiescent or time.monotonic() >= deadline:
1035:             break
1036:         # the drain is **bounded** and it never spins: the next attempt
1037:         # follows a short pause and it happens inside the same budget
1038:         time.sleep(0.01)
1039:     proof = _prove_salvage(all_salvage, prefix)
1040:     if proof is None and errors:
1041:         proof = '%s: the drain met %s' % (prefix, errors)
1042:     return {'owners': owners, 'hints': hints, 'salvage': salvage,
1043:             'all_salvage': all_salvage,
1044:             'attempts': attempts, 'proof': proof, 'errors': errors,
1045:             'quiescent': bool(quiescent)}
