398: class _PhaseLedger:
399:     """The ledger of the phases that the group of this module runs.
400: 
401:     The ledger remembers what was **entered** and what **ended**.  It
402:     never decides: a phase that was entered and that failed is a
403:     ``FAIL`` and only a phase that was never entered is a ``NOT_RUN``.
404:     The states are ``PLANNED``, ``ATTEMPTED``, ``PASSED``, ``FAIL`` and
405:     ``NOT_RUN`` and the two last states are terminal.
406:     """
407: 
408:     STATES = ('PLANNED', 'ATTEMPTED', 'PASSED', 'FAIL', 'NOT_RUN')
409: 
410:     def __init__(self):
411:         self._lock = threading.Lock()
412:         self._states = {}
413:         self._order = []
414:         self._reasons = {}
415: 
416:     def _plan_locked(self, name):
417:         if name not in self._states:
418:             self._states[name] = 'PLANNED'
419:             self._order.append(name)
420: 
421:     def plan(self, name):
422:         """Name one phase **before** it runs."""
423:         with self._lock:
424:             self._plan_locked(name)
425:             return name
426: 
427:     def attempt(self, name):
428:         """Mark that the operation of the phase was **entered**."""
429:         with self._lock:
430:             self._plan_locked(name)
431:             if self._states[name] == 'PLANNED':
432:                 self._states[name] = 'ATTEMPTED'
433:             return self._states[name]
434: 
435:     def finish(self, name, state, reason=None):
436:         """Close one phase; a closed phase is never reopened."""
437:         with self._lock:
438:             self._plan_locked(name)
439:             if self._states[name] in ('PASSED', 'FAIL', 'NOT_RUN'):
440:                 return self._states[name]
441:             self._states[name] = state
442:             if reason is not None:
443:                 self._reasons[name] = reason
444:             return state
445: 
446:     def pass_(self, name):
447:         """The phase held every one of its checks."""
448:         return self.finish(name, 'PASSED')
449: 
450:     def fail(self, name, reason=None):
451:         """The phase was executed and it failed."""
452:         return self.finish(name, 'FAIL', reason)
453: 
454:     def state(self, name):
455:         """The state of one phase."""
456:         with self._lock:
457:             return self._states.get(name, 'PLANNED')
458: 
459:     def names(self):
460:         """The names of the phases in the order of their planning."""
461:         with self._lock:
462:             return list(self._order)
463: 
464:     def reason(self, name):
465:         """The reason of the failure of one phase."""
466:         with self._lock:
467:             return self._reasons.get(name)

505: def _owner_start_state(thread, hint=None):
506:     """The start state of one owner of the round.
507: 
508:     The state is one of ``'STARTED'``, ``'NEVER_STARTED'`` or
509:     ``'UNKNOWN'``.  ``UNKNOWN`` is the state of an owner whose start
510:     did not return and of an owner whose liveness could not be asked:
511:     such an owner may still come up, it is never taken for an owner
512:     that never started and it **fails the proof** of the cleanup.
513: 
514:     CORRECTION 17 (M3): ``NEVER_STARTED`` is returned **only** when the
515:     provenance of the start **proves** it: the owner does not exist at
516:     all, or the registry of the fixture observed that its start was
517:     never invoked.  An owner without an identity and without a
518:     provenance is ``UNKNOWN``: it is never guessed to be an owner that
519:     never started.
520:     """
521:     if thread is None:
522:         return 'NEVER_STARTED'
523:     try:
524:         ident = getattr(thread, 'ident', None)
525:     except Exception:  # noqa: BLE001 - the failure stays visible
526:         return 'UNKNOWN'
527:     if ident is not None:
528:         return 'STARTED'
529:     if hint in ('STARTED', 'NEVER_STARTED', 'UNKNOWN'):
530:         return hint
531:     return 'UNKNOWN'

534: def _salvage_threads(owners, *, seconds=10.0, states=None, total_seconds=30.0):
535:     """Join the owners that really started and report their liveness.
536: 
537:     The salvage is the work of the **test** and it is **separate** from
538:     the returned state of the product: it never renews a deadline of
539:     the round, it never erases a retained resource of the product and
540:     it never claims that the product released it.
541: 
542:     Every owner that really started is joined with an explicitly
543:     bounded wait and its liveness is reported.  An owner that never
544:     started is reported and **never** joined.  An owner whose start
545:     state is ``UNKNOWN`` is reported as unknown: it is never joined as
546:     if it had ended and it fails the proof of the cleanup.
547: 
548:     CORRECTION 17 (M3): **every** owner is attempted.  A failure of the
549:     classification of an owner, of its join or of its liveness query
550:     never aborts the iteration: the error is **retained** in a list
551:     with its kind, its exact type and its detail, and it fails the
552:     proof of the cleanup.  The wait of an owner is cropped by the
553:     remaining budget of the salvage, so that the salvage is bounded as
554:     a whole.
555:     """
556:     facts = []
557:     hints = states or {}
558:     deadline = time.monotonic() + max(0.0, total_seconds)
559:     for name, thread in owners:
560:         state = None
561:         errors = []
562:         try:
563:             state = _owner_start_state(thread, hints.get(name))
564:         except BaseException as error:  # noqa: BLE001 - it stays visible
565:             errors.append({'kind': 'STATE', 'type': type(error).__name__,
566:                            'detail': str(error)})
567:             state = 'UNKNOWN'
568:         if state != 'STARTED':
569:             facts.append({'name': name, 'start_state': state,
570:                           'started': None if state == 'UNKNOWN' else False,
571:                           'alive': None, 'joined': None, 'errors': errors})
572:             continue
573:         try:
574:             wait = max(0.0, min(seconds, deadline - time.monotonic()))
575:             thread.join(wait)
576:         except BaseException as error:  # noqa: BLE001 - it stays visible
577:             errors.append({'kind': 'JOIN', 'type': type(error).__name__,
578:                            'detail': str(error)})
579:         alive = None
580:         try:
581:             alive = bool(thread.is_alive())
582:         except BaseException as error:  # noqa: BLE001 - it stays visible
583:             errors.append({'kind': 'LIVENESS', 'type': type(error).__name__,
584:                            'detail': str(error)})
585:         facts.append({'name': name, 'start_state': state, 'started': True,
586:                       'alive': alive, 'joined': (alive is False),
587:                       'errors': errors})
588:     return facts

793:             key = ('thread', id(thread)) if thread is not None \
794:                 else ('name', name)
795:             if key in seen:
796:                 continue
797:             seen.add(key)
798:             owners.append((name, thread))
799: 
800:     for entry in registry.entries():
801:         name = entry['name']
802:         owner = entry['owner']
803:         kind = entry['kind']
804:         kinds.add(kind)
805:         if kind == 'channel':
806:             add(_channel_owners(owner, name, hints))
807:         elif kind == 'endpoint':
808:             add(_endpoint_owners(name, owner, hints))
809:         elif kind == 'session':
810:             add(_session_owners(owner, name, hints))
811:         else:
812:             add([(name, owner)])

814:         add(_session_owners(session, '%s:session' % prefix, hints))
815:     if channel is not None and 'channel' not in kinds:
816:         add(_channel_owners(channel, '%s:channel' % prefix, hints))
817:     # CORRECTION 17 (M3): the provenance of the starts that the
818:     # observer of the fixture saw is the **proven** provenance of the
819:     # start of an owner.  It is keyed by the exact thread object and it
820:     # never replaces the hint of the production: a thread whose start
821:     # returned is ``STARTED`` and a thread whose start did not return is
822:     # ``UNKNOWN``.
823:     provenance = registry.start_provenance() if registry is not None else {}
824:     for name, thread in owners:
825:         if thread is None:
826:             continue

828:         if state == 'STARTED':

830:         elif state in ('STARTING', 'FAILED'):
831:             hints[name] = 'UNKNOWN'

835: def _drain_owners(registry, session=None, channel=None, prefix='round', *,
836:                   seconds=10.0, budget=30.0, release=None):
837:     """The bounded drain of the owners of one round.
838: 
839:     CORRECTION 17 (M3): the drain repeats, **boundedly**, until the
840:     owners of the round are quiescent and really joined.  The bound of
841:     the drain is **one** overall bound: the wait of an owner is cropped
842:     by the remaining budget and the budget is never reset - neither
843:     between the rounds of the drain nor between the rounds of the test.
844: 
845:     The quiescence of the round is asserted **after** the drain and
846:     never before it: a round that is not quiescent is a failure of the
847:     cleanup, the gates of the round are still released and **all** the
848:     registered owners are still attempted.
849:     """
850:     deadline = time.monotonic() + max(0.0, budget)
851:     attempts = []
852:     owners = []
853:     hints = {}
854:     salvage = []
855:     while True:
856:         owners, hints = _registry_owners(registry, session=session,
857:                                           channel=channel, prefix=prefix)
858:         snapshot = {'generation': registry.generation(),
859:                     'pending': registry.pending_starts(),
860:                     'owners': [name for name, _ in owners]}
861:         attempts.append(snapshot)
862:         if release is not None:
863:             release()
864:         remaining = max(0.0, deadline - time.monotonic())
865:         salvage = _salvage_threads(owners, states=hints, seconds=seconds,
866:                                    total_seconds=remaining)
867:         quiescent = (registry.pending_starts() == 0
868:                      and registry.generation() == snapshot['generation']
869:                      and _prove_salvage(salvage, prefix) is None)
870:         if quiescent or time.monotonic() >= deadline:
871:             break
872:         # the drain is **bounded** and it never spins: the next attempt
873:         # follows a short pause and it happens inside the same budget
874:         time.sleep(0.01)
875:     return {'owners': owners, 'hints': hints, 'salvage': salvage,
876:             'attempts': attempts, 'proof': _prove_salvage(salvage, prefix),
877:             'quiescent': (registry.pending_starts() == 0
878:                           and registry.generation() == attempts[-1]['generation'])}
