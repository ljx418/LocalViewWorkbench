694:     def job_process_ids(self, job):
695:         """The bounded list of the process ids of the job.
696: 
697:         The query is bounded by the descendant cap of this package.  A
698:         job that holds more processes than the cap is *not* reported as an
699:         empty job: the answer is unknown, and an unknown answer never
700:         becomes a zero.
701: 
702:         The structure of Windows holds two counts and an **inline** array
703:         of ``ULONG_PTR``.  The array is read from the buffer at the
704:         offset of the array and never through a pointer field.
705: 
706:         The answer is only accepted when **every** fact of it agrees:
707: 
708:         * the returned length is at least the size of the header, so that
709:           the two counts were really written and a legal empty job still
710:           carries its complete header;
711:         * the returned length is at most the size of the buffer, so that
712:           a length that is larger than the buffer was never accepted;
713:         * the returned length is at least the length that the count of
714:           the list needs, so that the list was not truncated;
715:         * the two counts are equal, so that the number of the assigned
716:           processes and the number of the listed processes agree;
717:         * the count of the list is within the cap of this package.
718: 
719:         Every other result is **unknown**: an exception, a returned
720:         length of zero, a reversed pair of counts, a count over the cap
721:         and a truncated list.  An unknown answer is never an empty job
722:         and never a success.
723:         """
724:         ctypes = self._api.types['ctypes']
725:         capacity = MAX_DESCENDANTS + 1
726:         pointer_size = ctypes.sizeof(ctypes.c_size_t)
727:         header_size = 2 * ctypes.sizeof(self._api.types['DWORD'])
728:         size = header_size + capacity * pointer_size
729:         buffer = ctypes.create_string_buffer(size)
730:         returned = self._api.types['DWORD'](0)
731:         if not self._api.kernel32.QueryInformationJobObject(
732:                 job, self.JOB_OBJECT_BASIC_PROCESS_ID_LIST, buffer, size,
733:                 ctypes.byref(returned)):
734:             error = self._last_error()
735:             if error == self.ERROR_MORE_DATA:
736:                 raise BrokerError('BROKER_DESCENDANT_LIMIT', 'last_error=%d' % error)
737:             raise BrokerError('BROKER_JOB_QUERY_UNKNOWN', 'last_error=%d' % error)
738:         header = ctypes.cast(buffer, ctypes.POINTER(
739:             self._api.JOBOBJECT_BASIC_PROCESS_ID_LIST)).contents
740:         assigned = int(header.NumberOfAssignedProcesses)
741:         listed = int(header.NumberOfProcessIdsInList)
742:         reported = int(returned.value)
743:         # The returned length has to carry the whole header and it may
744:         # never be larger than the buffer that was handed to Windows.
745:         if reported < header_size or reported > size:
746:             raise BrokerError('BROKER_JOB_QUERY_TRUNCATED',
747:                               'returned=%d:header=%d:allocated=%d'
748:                               % (reported, header_size, size))
749:         if assigned > capacity or listed > capacity:
750:             raise BrokerError('BROKER_DESCENDANT_LIMIT',
751:                               'assigned=%d:listed=%d' % (assigned, listed))
752:         if assigned != listed:

754:             # never an empty job.
755:             raise BrokerError('BROKER_JOB_QUERY_TRUNCATED',
756:                               'assigned=%d:listed=%d' % (assigned, listed))
757:         if listed > MAX_DESCENDANTS:
758:             raise BrokerError('BROKER_DESCENDANT_LIMIT',
759:                               'assigned=%d:listed=%d' % (assigned, listed))
760:         needed = header_size + listed * pointer_size
761:         if needed > reported:
762:             raise BrokerError('BROKER_JOB_QUERY_TRUNCATED',
763:                               'returned=%d:needed=%d' % (reported, needed))

1432:     def cleanup_deadline(self, deadline=None):
1433:         """The ONE absolute deadline of the cleanup of this round.
1434: 
1435:         The deadline is generated once, at the first stop or at the first
1436:         failure of the cleanup.  It is the earlier of the absolute
1437:         deadline of the round and the fixed cleanup budget, measured from
1438:         the moment of the first cleanup.  Every later call returns the
1439:         very same value: no layer of the cleanup adds a budget of its
1440:         own and no layer starts a new wait with a fresh timeout.
1441:         """
1442:         if self._cleanup_deadline_value is not None:
1443:             return self._cleanup_deadline_value
1444:         now = self._clock()
1445:         value = now + CLEANUP_BUDGET_SECONDS
1446:         if self._absolute_deadline is not None:
1447:             value = min(value, self._absolute_deadline)
1448:         if deadline is not None:
1449:             value = min(value, deadline)
1450:         self._cleanup_deadline_value = value
1451:         return value

1453:     def remaining_cleanup_seconds(self, deadline=None):
1454:         """The seconds that are left until the cleanup deadline.

1457:         consumed part of the budget never gets the whole budget again.
