 1125      def run_fault_scenario(run_root, process, phase):
 1126          """One no-channel scenario: the release, the facts, the recovery.
 1127  
 1128          The scenario runs on a **scoped** table of the launcher: the
 1129          table of the package is shared by the whole batch and the
 1130          context that another check of the batch bound to it is not the
 1131          context of this launch.  The production table is used and it is
 1132          never emptied by hand.
 1133  
 1134          RECONCILIATION (N1 correction 14, F1): the table of the launcher
 1135          converges **one release later** and not in the release that
 1136          closed the last original.  The reconciliation of the registered
 1137          originals of a release names the unresolved ones as **retained**
 1138          before the retry of the same release closes them: the report of
 1139          that release therefore names a retained name that is already
 1140          closed when the report is read.  The aggregate that the table
 1141          keeps is an **empty bookkeeping owner** - it has no process, no
 1142          channel and no original raw stream any more and it needs no
 1143          primitive work at all.
 1144  
 1145          The scenario proves that empty owner, records it honestly and
 1146          then permits **ONE** further explicit release of the **same**
 1147          owner under the **same** original positive deadline.  That
 1148          release performs **zero** close, terminate and wait and it
 1149          converges the table to the empty state.  The table is never
 1150          emptied by hand, the deadline is never renewed, the loop is
 1151          never repeated until the table is green and no check of a
 1152          retained resource is relaxed.
 1153          """
 1154          scoped = _LauncherCustody()
 1155          # The ONE absolute deadline of this scenario.  The first
 1156          # release, the recovery of the same owner and the final
 1157          # convergence use the **same** positive deadline: a release
 1158          # never opens a budget of its own and a convergence never
 1159          # renews a lease.
 1160          deadline = time.monotonic() + 30.0
 1161          with unittest.mock.patch.object(broker_module, '_LAUNCHER_CUSTODY',
 1162                                          scoped):
 1163              holder = m07a_interrupted_launch(run_root, sealed_plan(),
 1164                                               process=process)
 1165              assert holder['interrupted'] is True, \
 1166                  'the hash of the launch was not interrupted'
 1167              aggregate = holder['aggregate'][0] if holder['aggregate'] else None
 1168              assert aggregate is not None, \
 1169                  'the interrupted launch lost its owner'
 1170              cancellation = None
 1171              report = None
 1172              with unittest.mock.patch.object(broker_module, 'os',
 1173                                              FaultOsView()):
 1174                  try:
 1175                      report = scoped.release(deadline=deadline)
 1176                  except (KeyboardInterrupt, SystemExit) as error:
 1177                      # The **original** cancellation object is the error
 1178                      # that leaves.
 1179                      cancellation = error
 1180              first_facts = {'cancellation': (type(cancellation).__name__
 1181                                              if cancellation is not None
 1182                                              else None),
 1183                             'report': report,
 1184                             'deadline': deadline,
 1185                             'raw_streams': [(item['name'], repr(item['stream']))
 1186                                             for item in aggregate.raw_streams],
 1187                             'owned': sorted((item['kind'],
 1188                                              repr(item['resource']))
 1189                                             for item in scoped.items()),
 1190                             'process_owned': aggregate.process is process,

 1192                             'waited': process.waited,
 1193                             'handle_attempts': process._handle.attempts,
 1194                             'handle_closed': process._handle.closed,
 1195                             'stdin_attempts': process.stdin.attempts,
 1196                             'stdout_attempts': process.stdout.attempts}
 1197              write_phase('%s-first' % phase, first_facts)
 1198              # The recovery of the **same** owner: every unresolved
 1199              # original is really closed and the table converges.
 1200              with unittest.mock.patch.object(broker_module, 'os',
 1201                                              FaultOsView()):
 1202                  recovered = scoped.release(deadline=deadline)
 1203              recovered_facts = {'recovered': recovered,
 1204                             'deadline': deadline,
 1205                             'raw_streams': [(item['name'], repr(item['stream']))
 1206                                             for item in aggregate.raw_streams],
 1207                             'owned': sorted((item['kind'],
 1208                                              repr(item['resource']))
 1209                                             for item in scoped.items()),
 1210                             'stdin_attempts': process.stdin.attempts,
 1211                             'stdout_attempts': process.stdout.attempts,
 1212                             'stdin_closed': process.stdin.closed,
 1213                             'stdout_closed': process.stdout.closed,
 1214                             'handle_attempts': process._handle.attempts,
 1215                             'handle_closed': process._handle.closed,
 1216                             'terminated': process.terminated,
 1217                             'waited': process.waited}
 1218              write_phase('%s-recovery' % phase, recovered_facts)
 1219              # -- the state of the table **before** the salvage: the
 1220              #    table may keep **only** the empty bookkeeping owner of
 1221              #    the aggregate.  Every original of the launch was really
 1222              #    closed and the name that is kept is the name of a
 1223              #    snapshot of an earlier release of the same owner.
 1224              retained_items = scoped.items()
 1225              empty_facts = {
 1226                  'owned': sorted((item['kind'], repr(item['resource']))
 1227                                  for item in retained_items),
 1228                  'process_is_none': aggregate.process is None,
 1229                  'channel_is_none': aggregate.channel is None,
 1230                  'raw_streams': [item['name'] for item in aggregate.raw_streams],
 1231                  'workers': len(aggregate.workers),
 1232                  'endpoints': len(aggregate.endpoints),
 1233                  'stdin_closed': process.stdin.closed,
 1234                  'stdout_closed': process.stdout.closed,
 1235                  'stdin_confirmed': bool(
 1236                      aggregate.progress.fact('stream:stdin').get('closed')),
 1237                  'stdout_confirmed': bool(
 1238                      aggregate.progress.fact('stream:stdout').get('closed')),
 1239                  'handle_attempts': process._handle.attempts,
 1240                  'handle_closed': process._handle.closed,
 1241                  'terminated': process.terminated,
 1242                  'waited': process.waited,
 1243              }
 1244              write_phase('%s-empty-owner' % phase, empty_facts)
 1245              assert [item['kind'] for item in retained_items] in ([], ['aggregate']), \
 1246                  'the table of the launcher kept an original resource of the launch'
 1247              assert aggregate.process is None, \

 1254                  'the aggregate of the launch still owns a live worker or endpoint'
 1255              assert process.stdin.closed is True, \
 1256                  'the original input of the launch was not really closed'
 1257              assert process.stdout.closed is True, \
 1258                  'the original output of the launch was not really closed'
 1259              assert empty_facts['stdin_confirmed'] is True, \
 1260                  'the close of the original input was not confirmed by its owner'
 1261              assert empty_facts['stdout_confirmed'] is True, \
 1262                  'the close of the original output was not confirmed by its owner'
 1263              assert process._handle.closed == 1, \
 1264                  'the original handle of the process was not closed exactly once'
 1265              # -- ONE further explicit release of the **same** owner
 1266              #    under the **same** deadline.  It converges the table
 1267              #    and it performs **zero** primitive work: no close, no
 1268              #    terminate and no wait.
 1269              with unittest.mock.patch.object(broker_module, 'os',
 1270                                              FaultOsView()):
 1271                  converged = scoped.release(deadline=deadline)
 1272              converged_facts = {'converged': converged,
 1273                             'deadline': deadline,
 1274                             'owned': sorted((item['kind'],
 1275                                              repr(item['resource']))
 1276                                              for item in scoped.items()),
 1277                             'stdin_attempts': process.stdin.attempts,
 1278                             'stdout_attempts': process.stdout.attempts,
 1279                             'stdin_closed': process.stdin.closed,
 1280                             'stdout_closed': process.stdout.closed,
 1281                             'handle_attempts': process._handle.attempts,
 1282                             'handle_closed': process._handle.closed,
 1283                             'terminated': process.terminated,
 1284                             'waited': process.waited}
 1285              write_phase('%s-converge' % phase, converged_facts)
 1286              # The table of the launcher converges: every original of

 1288              assert scoped.items() == [], \
