  189 def _started_owner(thread):
  190     """Whether the owner really started.
  191 
  192     A thread whose start did not return and that never ran has **no**
  193     identity: it is not a thread of the round and it is never joined.
  194     No exit is ever inferred from such an object.
  195     """
  196     if thread is None:
  197         return False
  198     if getattr(thread, 'ident', None) is not None:
  199         return True
  200     try:
  201         return bool(thread.is_alive())

  203         return False

  206 def _salvage_threads(owners, *, seconds=10.0):
  207     """Join the owners that really started and report their liveness.
  208 
  209     The salvage is the work of the **test** and it is **separate** from
  210     the returned state of the product: it never renews a deadline of
  211     the round, it never erases a retained resource of the product and
  212     it never claims that the product released it.
  213 
  214     Every owner that really started is joined with an explicitly
  215     bounded wait and its liveness is reported.  An owner that never
  216     started is reported and **never** joined.
  217     """
  218     facts = []
  219     for name, thread in owners:
  220         if not _started_owner(thread):
  221             facts.append({'name': name, 'started': False, 'alive': None,
  222                           'joined': None})
  223             continue
  224         try:
  225             thread.join(seconds)
  226         except RuntimeError:  # pragma: no cover - a thread of another round
  227             facts.append({'name': name, 'started': True, 'alive': True,
  228                           'joined': False})
  229             continue
  230         alive = bool(thread.is_alive())
  231         facts.append({'name': name, 'started': True, 'alive': alive,
  232                       'joined': not alive})
  233     return facts

  236 def _session_owners(session):
  237     """The owners that the session of the round really started.
  238 
  239     The registry is read from the session itself: the background
  240     threads of the round and the closer threads that it really built.
  241     A closer that never started is reported as such and it is never
  242     joined.
  243     """
  244     owners = []
  245     if session is None:
  246         return owners
  247     records = getattr(session, '_background_threads', None) or {}
  248     for kind in sorted(records):
  249         record = records[kind] or {}
  250         owners.append(('background:' + str(kind), record.get('thread')))
  251     for index, thread in enumerate(getattr(session, '_closer_threads', None)
  252                                    or []):
  253         owners.append(('closer:%d' % index, thread))
  254     return owners

  272                        getattr(reader, '_reader', None)))
  273     if writer is not None:
  274         owners.append((prefix + ':write-worker',
  275                        getattr(writer, '_writer', None)))
